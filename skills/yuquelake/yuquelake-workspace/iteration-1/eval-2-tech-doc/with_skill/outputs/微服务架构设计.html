核心算法示例

def calculate_refund(balance, rebate, discount=0.75): """计算退款金额""" base = (balance - rebate) * discount fee = min(balance * 0.2, 100) return base - fee result = calculate_refund(1000, 0) print(f"退款金额: {result}")
